\ifnum#2=0
  \ifnum#1=1
    \draw (0,3.7)--(0,3.3) to[nos,l=$A$] (0,2)--(0,1.8) to[nos,l=$B$] (0,.5)--(0,.1) node[ground]{};
  \fi
  \ifnum#1=2
    \draw (0,3.7)--(0,3.3) to[D,l=$D_1$] (0,2)--(0,1.8) to[D,l=$D_2$] (0,.5)--(0,.1) node[ground]{};
  \fi
  \ifnum#1=3
    \node[nmos] (a) at (0,2.65) {};\node[nmos] (b) at (0,1.15) {};
    \draw (0,3.7)--(a.D);\draw (a.S)--(b.D);\draw (b.S)--(0,.1) node[ground]{};
    \draw[blue_theme,thick] (a.G)--++(-.5,0) node[left]{$A$};\draw[blue_theme,thick] (b.G)--++(-.5,0) node[left]{$B$};
  \fi

  \draw (0,3.7)--(0,3.2)--(-1.1,3.2) (0,3.2)--(1.1,3.2);

    \draw (-1.1,3.2) to[nos,l=$A$] (-1.1,.8);\draw (1.1,3.2) to[nos,l=$B$] (1.1,.8);
  \fi

    \draw (-1.1,3.2) to[D,l=$D_1$] (-1.1,.8);\draw (1.1,3.2) to[D,l=$D_2$] (1.1,.8);
  \fi

    \draw[blue_theme,thick] (a.G)--++(-.25,0) node[left]{$A$};\draw[blue_theme,thick] (b.G)--++(-.25,0) node[left]{$B$};
  \fi
  \draw (-1.1,.8)--(1.1,.8);\draw (0,.8)--(0,.1) node[ground]{};
