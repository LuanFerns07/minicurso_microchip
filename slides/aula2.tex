\section{NAND}
\begin{frame}{Série ou paralelo: quando existe caminho?}
\duas{\centering\textcolor{blue_theme}{\large\textbf{Série}}\par
\figura{\begin{circuitikz}[scale=.8]\node[nmos] (a) at(0,1){};\node[nmos] (b) at(0,-.6){};\draw(a.D)--(0,1.8) node[above]{$Y$};\draw(a.S)--(b.D);\draw(b.S)--(0,-1.4) node[ground]{};\draw(a.G) node[left]{$A$};\draw(b.G) node[left]{$B$};\end{circuitikz}}

\figura{\begin{circuitikz}[scale=.8]\node[nmos](a) at(-1,0){};\node[nmos](b) at(1.2,0){};\draw(a.D)--(-1,1)--(1.2,1)--(b.D);\node[above] at(0,1){$Y$};\draw(a.S)--(-1,-1)--(1.2,-1)--(b.S);\draw(0,-1)node[ground]{};\draw(a.G)node[left]{$A$};\draw(b.G)node[left]{$B$};\end{circuitikz}}

\end{frame}
\note{As frases descrevem a condução até GND, não o valor lógico na saída. Quando um caminho de descida conduz, a saída é levada a zero. Uma rede NMOS isolada não garante o nível alto. Usar dedos sobre o desenho para mostrar interrupções.}

\begin{frame}{NAND: descarregar somente com A e B altos}
\duas{\figura{\nandcircuito}}{\vspace{8mm}\[Y=\overline{AB}\]\par\bigskip
NMOS: série.\par\medskip
PMOS: paralelo.\par\medskip
Quatro transistores.}
\end{frame}
\note{Começar pela condição Y=0: A e B precisam estar altos, portanto a descida é uma série. Para qualquer entrada baixa, deve existir um caminho de subida: PMOS em paralelo. As etiquetas A e B indicam as mesmas entradas nas portas correspondentes. Corpos omitidos, ligados aos trilhos.}

\begin{frame}{NAND: mude uma entrada e siga a carga}
\only<1|handout:0>{\figura{\portaestado{0}{0}{0}}\etapa{1/4}{00: os dois PMOS oferecem caminhos de subida.}}
\only<2|handout:0>{\figura{\portaestado{0}{0}{1}}\etapa{2/4}{01: A interrompe a série NMOS; PMOS A mantém Y alto.}}
\only<3|handout:0>{\figura{\portaestado{0}{1}{1}}\etapa{3/4}{11: a série NMOS está completa; Y desce.}}
\only<4|handout:1>{\figura{\portaestado{0}{1}{0}}\etapa{4/4}{10: B interrompe a série; PMOS B faz Y subir.}}
\end{frame}
\note{A ordem 00, 01, 11, 10 muda um bit por vez. Verde identifica um dispositivo habilitado no modelo de chave; só um caminho completo até um trilho fixa a saída. Uma chave verde dentro de uma série interrompida não descarrega sozinha a saída. Em pilhas reais, correntes e tensões internas exigem o modelo elétrico. Pedir a previsão antes de cada clique.}

\section{NOR}
\begin{frame}{NOR: qualquer entrada alta faz a saída descer}
\duas{\figura{\norcircuito}}{\vspace{8mm}\[Y=\overline{A+B}\]\par\bigskip
NMOS: paralelo.\par\medskip
PMOS: série.\par\medskip
Quatro transistores.}
\end{frame}
\note{Dois caminhos alternativos de descida: um comandado por A e outro por B. A subida precisa dos dois PMOS habilitados, portanto é uma série. Comparar com NAND sem entregar apenas uma regra para decorar.}

\begin{frame}{NOR: observe a série PMOS}
\only<1|handout:0>{\figura{\portaestado{1}{0}{0}}\etapa{1/4}{00: a série PMOS está completa; Y sobe.}}
\only<2|handout:0>{\figura{\portaestado{1}{0}{1}}\etapa{2/4}{01: NMOS B descarrega; PMOS B interrompe a subida.}}
\only<3|handout:0>{\figura{\portaestado{1}{1}{1}}\etapa{3/4}{11: os dois ramos NMOS descarregam a saída.}}
\only<4|handout:1>{\figura{\portaestado{1}{1}{0}}\etapa{4/4}{10: basta o ramo NMOS A para manter Y baixo.}}
\end{frame}
\note{Verde indica dispositivo habilitado no modelo ideal, e não um certificado de caminho completo. A série PMOS precisa dos dois dispositivos: em 01 e 10 há um PMOS habilitado, mas nenhuma subida completa. Pedir que a turma descreva o caminho que existe e o que falta.}

\begin{frame}{Agora confira a tabela da NOR}

\end{frame}
\note{Comparar a linha 00 com 11 da NAND. Corrigir a ideia de que basta olhar o número de transistores: as duas portas têm quatro, mas sua topologia e função são diferentes.}

\begin{frame}{Mesmo material, outra organização}
\duas{\centering\textbf{NAND}\par\figura{\nandcircuito}\small Série embaixo; paralelo em cima.}{\centering\textbf{NOR}\par\figura{\norcircuito}\small Paralelo embaixo; série em cima.}
\end{frame}
\note{Usar o quadro como consulta visual para a atividade. Na NAND, saída zero apenas em 11; na NOR, saída um apenas em 00. Área, corrente e atraso não são determinados unicamente pela contagem de transistores.}

\begin{frame}{De Morgan: a razão da troca}
\[\overline{AB}=\overline A+\overline B\qquad\overline{A+B}=\overline A\,\overline B\]
\figura{\begin{tikzpicture}[node distance=10mm,font=\small]

\end{tikzpicture}}
\chamada{Mesmas entradas nas portas; redes duais.}
\end{frame}
\note{O PMOS conduz com nível baixo; as negações surgem desse comportamento, sem acrescentar inversores às portas. A receita série/paralelo é para CMOS estático complementar. Funções mais gerais podem exigir entradas complementadas ou composição de células.}

\section{Composição}
\begin{frame}{Uma inversão a mais produz AND ou OR}
\only<1|handout:0>{\figura{\begin{circuitikz}\node[nand port](n)at(0,0){};\node[not port](i)at(3,0){};\draw(n.in 1)--++(-.5,0)node[left]{$A$};\draw(n.in 2)--++(-.5,0)node[left]{$B$};\draw(n.out)--node[above]{$\overline{AB}$}(i.in);\draw(i.out)--++(.6,0)node[right]{$AB$};\end{circuitikz}}\chamada{NAND + inversor = AND.}}
\only<2|handout:1>{\figura{\begin{circuitikz}\node[nor port](n)at(0,0){};\node[not port](i)at(3,0){};\draw(n.in 1)--++(-.5,0)node[left]{$A$};\draw(n.in 2)--++(-.5,0)node[left]{$B$};\draw(n.out)--node[above]{$\overline{A+B}$}(i.in);\draw(i.out)--++(.6,0)node[right]{$A+B$};\end{circuitikz}}\chamada{NOR + inversor = OR.}}
\end{frame}
\note{São duas células, com nó intermediário e carga entre estágios. Cada implementação didática usa 4+2=6 transistores. O quadro de consulta resume as duas construções. Não afirmar que são as únicas topologias possíveis nem as de menor área.}

\begin{frame}{Uma NAND também pode inverter}
\figura{\begin{circuitikz}\node[nand port](n)at(1.7,0){};\draw(n.in 1)--(-.2,.28)--(-.2,-.28)--(n.in 2);\draw(-.2,0)to[short,*-](-1,0)node[left]{$A$};\draw(n.out)--++(.9,0)node[right]{$\overline A$};\end{circuitikz}}
\[N(A,A)=\overline{AA}=\overline A\]
\chamada{Compondo NANDs, construímos qualquer função booleana.}
\end{frame}
\note{A união das entradas deve ser desenhada de forma clara. A universalidade segue de obter NOT, AND e OR com NANDs. Mostrar a OR por De Morgan na lousa se a turma pedir; evitar três expressões aninhadas na tela. NOR também é universal. Um inversor dedicado usa dois transistores; a NAND com entradas unidas continua usando sua célula de quatro.}

\begin{frame}{O que vamos levar para a prática?}
\begin{enumerate}\item Prever a tabela verdade pelo desenho.
\item Identificar o caminho de carga e de descarga.
\item Comparar a previsão com as formas de onda.\end{enumerate}
\chamada{O circuito deve fazer sentido antes da simulação.}
\end{frame}
\note{Ponte conceitual com a futura prática no Virtuoso. O roteiro de laboratório, comandos, PDK e dimensões serão elaborados posteriormente. As formas de onda poderão mostrar atrasos e efeitos de carga que o modelo lógico não calcula.}

\begin{frame}{Saída da oficina: explique uma escolha}
\begin{enumerate}\item Por que dois NMOS em série levam à NAND na saída?
\item Em uma NOR com entradas 01, quem descarrega?
\item O que muda eletricamente quando uma porta comuta?\end{enumerate}
\chamada{Uma resposta, um caminho no desenho.}
\end{frame}
\note{Respostas: a condição AND ativa a descarga, negando o valor na saída; o NMOS B; a carga das capacitâncias muda, levando tempo e consumindo energia. Retomar dúvidas da atividade antes de encerrar.}
