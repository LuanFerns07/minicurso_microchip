\section{NAND}
\begin{frame}{Da resistência R à rede complementar}
\only<1|handout:0>{\duas{\figura{\construirNAND{1}}}{\vspace{10mm}\textcolor{blue_theme}{\large\textbf{O ponto de partida}}\par\medskip
Dois NMOS em série.\par\medskip R fornece o caminho de subida.\par\medskip
Com A=B=1, há corrente por R.}\etapa{1/2}{A saída baixa mantém um caminho entre alimentação e GND.}}
\only<2|handout:1>{\duas{\figura{\construirNAND{2}}}{\vspace{10mm}\textcolor{blue_theme}{\large\textbf{Construção CMOS}}\par\medskip
Substituímos R por dois PMOS em paralelo.\par\medskip
Eles conduzem quando A=0 ou B=0.}\etapa{2/2}{A rede de subida bloqueia quando a série NMOS conduz.}}
\fonte{Entradas binárias estáveis; modelo de chaves ideais. Os Gates A/B são compartilhados pelas duas redes. Corpos: NMOS em GND, PMOS em VDD, no modelo bulk.}
\end{frame}

\begin{frame}{Uma rede conduz; a outra precisa bloquear}
\duas{\figura{\portaestado{0}{1}{1}}}{\vspace{8mm}\textcolor{blue_theme}{\large\textbf{A=B=1}}\par\medskip
NMOS: caminho completo até GND.\par\medskip PMOS: dois caminhos interrompidos.\par\medskip
Logo, $Y=0$.}
\chamada{Condição de condução $AB$; saída invertida $\overline{AB}$.}
\fonte{Verde indica transistor habilitado. Uma série só conduz por completo quando todos os seus dispositivos estão habilitados.}
\end{frame}

\begin{frame}{NAND: descarregar somente com A e B altos}
\duas{\figura{\nandcircuito}}{\vspace{8mm}\[Y=\overline{AB}\]\par\bigskip
NMOS: série.\par\medskip
PMOS: paralelo.\par\medskip
Quatro transistores.}
\end{frame}

\begin{frame}{NAND: mude uma entrada e siga a carga}
\only<1|handout:0>{\figura{\portaestado{0}{0}{0}}\etapa{1/4}{00: os dois PMOS oferecem caminhos de subida.}}
\only<2|handout:0>{\figura{\portaestado{0}{0}{1}}\etapa{2/4}{01: A interrompe a série NMOS; PMOS A mantém Y alto.}}
\only<3|handout:0>{\figura{\portaestado{0}{1}{1}}\etapa{3/4}{11: a série NMOS está completa; Y desce.}}
\only<4|handout:1>{\figura{\portaestado{0}{1}{0}}\etapa{4/4}{10: B interrompe a série; PMOS B faz Y subir.}}
\fonte{Verde: transistor habilitado. Siga um caminho completo entre Y e uma das alimentações.}
\end{frame}

\begin{frame}{Dois erros que a complementaridade evita}

\begin{frame}{NOR: observe a série PMOS}

\begin{frame}{De Morgan: por que série vira paralelo?}
\figura{\redesDuais}
\only<1|handout:0>{\etapa{1/2}{NMOS habilita com 1; PMOS habilita com 0.}}
\only<2|handout:1>{\etapa{2/2}{$\overline{AB}=\overline A+\overline B$ e $\overline{A+B}=\overline A\,\overline B$.}}
\fonte{As expressões nas redes representam condições de condução. Trocar série por paralelo constrói a rede dual; os Gates mantêm os mesmos sinais A/B.}
\end{frame}
