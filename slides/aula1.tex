\begin{frame}{Duas chaves complementares}
\duas{\centering\textcolor{blue_theme}{\large\textbf{NMOS}}\par
\figura{\begin{circuitikz}\node[nmos] (n) at (0,0) {};\draw (n.G) node[left]{G};\draw(n.S) node[below]{S};\draw(n.D) node[above]{D};\end{circuitikz}}
\[V_{GS}=V_G-V_S\]
Conduz se $V_{GS}>V_{TN}$.}{\centering\textcolor{blue_theme}{\large\textbf{PMOS}}\par
\figura{\begin{circuitikz}\node[pmos] (p) at (0,0) {};\draw(p.G) node[left]{G};\draw(p.S) node[above]{S};\draw(p.D) node[below]{D};\end{circuitikz}}
\[V_{SG}=V_S-V_G\]
Conduz se $V_{SG}>|V_{TP}|$.}
\chamada{Compare terminais, não apenas a tensão de Gate.}
\fonte{Modelo ideal de chave para MOSFETs de enriquecimento; a corrente real não se anula abruptamente no limiar.}
\end{frame}

\begin{frame}{Qual chave está ligada?}
\centering\small
\only<1|handout:0>{\begin{tabular}{llll}\toprule Tipo & Gate & Source & Estado\\\midrule NMOS&0 V&0 V&?\\PMOS&0,2 V&1,2 V&?\\NMOS&1,2 V&1,0 V&?\\\bottomrule\end{tabular}\par\bigskip Calcule $V_{GS}$ ou $V_{SG}$ antes de responder.}
\only<2|handout:1>{\begin{tabular}{llll}\toprule Tipo & Diferença & Comparação & Estado\\\midrule NMOS&$V_{GS}=0$ V&$0<0{,}4$&Cortado\\PMOS&$V_{SG}=1{,}0$ V&$1{,}0>0{,}4$&Ligado\\NMOS&$V_{GS}=0{,}2$ V&$0{,}2<0{,}4$&Cortado\\\bottomrule\end{tabular}\chamada{Gate alto não basta se o Source também estiver alto.}}
\fonte{Valores didáticos: limiares constantes $V_{TN}=|V_{TP}|=0{,}4$ V. Não são parâmetros do laboratório.}
\end{frame}
