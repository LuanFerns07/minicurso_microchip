\section{Semicondutores}
\begin{frame}{Antes do silício: tensão, corrente e caminho}
\duas{\topico{Tensão}{Diferença de potencial entre dois pontos.\\[2mm]$V_{AB}=V_A-V_B$\\[2mm]Toda tensão precisa de uma referência.}
\topico{Corrente}{Fluxo de carga em um caminho.\\[2mm]$I=\Delta Q/\Delta t$}}{\topico{Chave ideal}{Fechada: oferece um caminho de baixa resistência.\\[2mm]Aberta: interrompe o caminho; não fixa a tensão dos terminais.}
\topico{Capacitância}{Armazena carga: $Q=CV$.\\Os nós de um circuito têm capacitâncias.}}
\end{frame}
\note{Uma capacitância inicialmente carregada pode manter tensão mesmo quando desconectada. O MOSFET real não é uma chave ideal.}

\begin{frame}{Condutor, isolante e semicondutor}
\centering\small\begin{adjustbox}{max width=\linewidth}\begin{tabular}{p{2.4cm}p{4.8cm}p{4.9cm}}\toprule Material & Comportamento qualitativo & Exemplo\\\midrule
Condutor & Muitos portadores disponíveis para condução & Cobre em interconexões\\[2mm]
Isolante & Poucos portadores disponíveis nas condições usuais & Dielétrico entre porta e canal\\[2mm]
Semicondutor & Condutividade sensível a dopagem, temperatura e campos & Silício no transistor\\\bottomrule\end{tabular}\end{adjustbox}
\vspace{3mm}\ideia{O interesse está em controlar a condução, e não apenas em conduzir bem.}
\end{frame}
\note{Classificação qualitativa. Distinguir o papel dos três materiais no mesmo circuito. O silício intrínseco não é apenas um material de resistência intermediária.}

\begin{frame}{Uma lacuna também participa da condução}
\begin{adjustbox}{max width=\linewidth}\begin{tikzpicture}[x=1.15cm,y=1cm]

\end{tikzpicture}\end{adjustbox}
\vspace{4mm}\begin{itemize}\item Elétron: portador de carga negativa.
\item Lacuna: ausência de um elétron na banda de valência; comporta-se como portador positivo.
\item O movimento equivalente da lacuna é oposto ao dos elétrons que a preenchem.\end{itemize}
\end{frame}
\note{Desenho de ocupação de estados, sem representar órbitas atômicas. Não confundir movimento eletrônico com sentido convencional da corrente.}

\begin{frame}{Silício intrínseco: geração e recombinação}

\begin{itemize}
\item No silício puro, a geração térmica cria um elétron livre e uma lacuna simultaneamente.
\item Na recombinação, um elétron ocupa um estado vazio e o par deixa de contribuir como portadores livres.
\item Em equilíbrio térmico, geração e recombinação se compensam em média.
\end{itemize}
\begin{block}{Material intrínseco}As concentrações de elétrons e lacunas são iguais: $n=p=n_i$. A temperatura influencia a quantidade de portadores disponíveis.\end{block}
\pergunta{Como aumentar preferencialmente uma dessas populações sem depender apenas da temperatura?}
\end{frame}
\note{A pergunta conduz à dopagem. n e p são concentrações, não cargas líquidas; ni é a concentração intrínseca. Não é necessário introduzir valores numéricos ou a estatística de Fermi neste nível.}


\begin{frame}{Dopagem: controlar a população de portadores}
\duas{\topico{Tipo N · doadores}{Impurezas como fósforo fornecem elétrons adicionais.\\[2mm]\textbf{Majoritários: elétrons.}\\Minoritários: lacunas.}}{\topico{Tipo P · aceitadores}{Impurezas como boro favorecem lacunas.\\[2mm]\textbf{Majoritários: lacunas.}\\Minoritários: elétrons.}}
\vspace{3mm}\ideia{Material tipo N não significa material com carga líquida negativa. O volume dopado permanece aproximadamente neutro.}
\end{frame}
\note{A neutralidade envolve cargas móveis e íons dopantes fixos. Não associar automaticamente dopagem do corpo à condução do canal MOS.}

\begin{frame}{Dois mecanismos de transporte de carga}

\topico{Deriva}{Um campo elétrico orienta o movimento dos portadores. Elétrons e lacunas se deslocam em sentidos opostos; suas contribuições à corrente convencional podem ter o mesmo sentido.}
\topico{Difusão}{Uma diferença de concentração leva portadores das regiões mais concentradas para as menos concentradas. Não é necessário aplicar uma tensão externa para iniciar esse processo.}
\ideia{Na junção PN, a difusão inicial cria um campo interno. Em equilíbrio, as correntes de deriva e difusão se compensam.}
\end{frame}
\note{Distinguir mecanismo de transporte de sentido da corrente. Não afirmar que todos os elétrons se movem alinhados ao campo: há movimento térmico e uma componente média de deriva. Usar esse quadro para explicar a formação da depleção.}

\item Íons fixos expostos criam campo elétrico e uma barreira de potencial.\end{itemize}
\end{frame}
\note{Em equilíbrio, deriva e difusão se compensam: corrente líquida nula. As cargas desenhadas são íons fixos na depleção. O campo aponta de N positivo para P negativo.}

\section{MOSFET}
\begin{frame}{MOSFET: quatro terminais e um canal controlável}
\centering\begin{adjustbox}{max width=\linewidth}\begin{tikzpicture}[x=.9cm,y=.65cm,font=\small]
\fill[blue_theme!8,draw=blue_theme] (0,0) rectangle (10,2.5);
\fill[yellow!40,draw=blue_theme] (.7,1.4) rectangle (2.4,2.5);
\fill[yellow!40,draw=blue_theme] (7.6,1.4) rectangle (9.3,2.5);
\fill[gray!30,draw=blue_theme] (2.7,2.55) rectangle (7.3,2.8);
\fill[blue_theme!60,draw=blue_theme] (2.7,2.8) rectangle (7.3,3.3);
\node at (1.55,1.9) {$n^+$};\node at (8.45,1.9) {$n^+$};\node at (5,.7) {Corpo P (B)};
\draw (1.55,2.5)--(1.55,3.8) node[above]{Fonte (S)};
\draw (8.45,2.5)--(8.45,3.8) node[above]{Dreno (D)};
\draw (5,3.3)--(5,3.8) node[above]{Porta (G)};
\draw (7.3,2.67)--(10.5,2.67) node[right]{Dielétrico};
\draw[dashed,thick,red] (2.4,2.3)--(7.6,2.3);
\node[red] at (5,1.6) {Região do canal};
\end{tikzpicture}\end{adjustbox}
\fonte{NMOS planar de enriquecimento, sem escala. Redesenho didático: Dutertre, fig. 1--3; Salem, p. 2.13--2.14.}
\end{frame}
\note{MOS: metal--óxido--semicondutor; FET: transistor de efeito de campo. O modelo planar não descreve todas as tecnologias. Não há contato condutor direto entre porta e canal.}

\begin{frame}{A porta controla por campo elétrico}
\begin{enumerate}\item Aplicamos uma tensão entre porta e fonte.
\item O campo através do dielétrico altera as cargas na superfície do corpo, sem exigir um contato condutor entre porta e canal.
\item No NMOS, uma tensão suficientemente positiva forma um canal de elétrons.
\item Com tensão entre dreno e fonte, pode circular corrente pelo canal.\end{enumerate}
\vspace{2mm}\ideia{A porta controla o canal. A corrente de condução principal circula entre dreno e fonte.}
\end{frame}
\note{Na formação do canal NMOS há depleção e inversão da superfície P. Corrente contínua de porta é pequena, mas capacitâncias exigem corrente nas transições. MOSFET não equivale a dois diodos em série.}

\begin{frame}{Da superfície P ao canal de elétrons}

Considere um NMOS com corpo P e aumente gradualmente a tensão de porta em relação ao corpo.
\begin{enumerate}
\item \textbf{Acumulação:} com porta suficientemente negativa, lacunas se concentram na superfície.
\item \textbf{Depleção:} uma tensão positiva afasta lacunas, expondo íons fixos próximos à superfície.
\item \textbf{Inversão:} para tensão positiva suficiente, elétrons formam um canal junto ao dielétrico.
\end{enumerate}
\ideia{A superfície pode passar a conduzir por elétrons mesmo que o corpo permaneça do tipo P.}
\end{frame}
\note{Modelo qualitativo de um MOS planar com corpo referenciado. O limiar marca uma transição convencional para inversão forte; não é uma fronteira física de corrente exatamente nula. A porta controla o canal sem transformar todo o substrato em material N.}


\begin{frame}{NMOS: comparar porta e fonte}
\duas{\centering\begin{adjustbox}{max width=\linewidth}\begin{circuitikz}[scale=1.1]
\node[nmos] (n) at (0,0) {};
\draw (n.D)--(0,1.2) node[above]{D};\draw (n.S)--(0,-1.2) node[below]{S};\draw (n.G)--(-1.6,0) node[left]{G};
\node[right] at (.4,0) {$V_{GS}=V_G-V_S$};
\end{circuitikz}\end{adjustbox}}{\topico{Modelo de chave}{\textbf{Conduz}: $V_{GS}>V_{TN}$.\\[2mm]\textbf{Cortado}: $V_{GS}<V_{TN}$, idealizando a fuga.}
\topico{Na rede de descida CMOS}{Com fonte próxima de GND, entrada alta tende a ligar o NMOS.}}
\fonte{Símbolo simplificado. Corpo NMOS ligado a GND no CMOS bulk introdutório.}
\end{frame}
\note{Modelo de enriquecimento em inversão forte. Há corrente de sub-limiar abaixo do limiar. Se a fonte elevar, o mesmo VG não garante a mesma condução; efeito de corpo fica para estudo posterior.}

\begin{frame}{PMOS: comparar fonte e porta}
\duas{\centering\begin{adjustbox}{max width=\linewidth}\begin{circuitikz}[scale=1.1]
\node[pmos] (p) at (0,0) {};
\draw (p.S)--(0,1.2) node[above]{S};\draw (p.D)--(0,-1.2) node[below]{D};\draw (p.G)--(-1.6,0) node[left]{G};
\node[right] at (.4,0) {$V_{SG}=V_S-V_G$};
\end{circuitikz}\end{adjustbox}}{\topico{Modelo de chave}{\textbf{Conduz}: $V_{SG}>|V_{TP}|$.\\[2mm]\textbf{Cortado}: $V_{SG}<|V_{TP}|$.}
\topico{Na rede de subida CMOS}{Com fonte próxima de $\VDD$, entrada baixa tende a ligar o PMOS.}}
\fonte{Símbolo com círculo na porta. Corpo/poço N do PMOS ligado a $\VDD$ neste modelo.}
\end{frame}
\note{VSG positivo evita decorar desigualdades com VTP negativo. PMOS tem difusões P em poço N. Não generalizar a regra de nível de porta sem considerar a fonte.}

\begin{frame}{O quarto terminal: por que ligar o corpo?}

\begin{itemize}
\item As regiões de fonte e dreno formam junções PN com o corpo ou o poço onde foram construídas.
\item No modelo CMOS bulk introdutório, o corpo do NMOS é ligado ao potencial mais baixo, GND.
\item O corpo/poço N do PMOS é ligado ao potencial mais alto, $\VDD$.
\item Isso ajuda a manter as junções sem polarização direta nos estados usuais do circuito.
\end{itemize}
\topico{Corpo e limiar}{A tensão entre fonte e corpo pode alterar o limiar. Nos exercícios iniciais, vamos idealizar esse efeito e usar limiares constantes.}
\end{frame}
\note{Não confundir corpo com fonte. Em transistores empilhados, a fonte de um dispositivo pode estar em um nó interno e seu corpo continuar no trilho. As conexões descritas são do modelo bulk convencional, não uma regra para qualquer processo ou estrutura.}


\begin{frame}{Exemplo numérico · o que realmente liga a chave?}
\small\centering\begin{adjustbox}{max width=\linewidth}\begin{tabular}{llll}\toprule Dispositivo & Tensões & Comparação & Estado ideal\\\midrule
NMOS & $V_G=1{,}2$ V; $V_S=0$ V & $V_{GS}=1{,}2>0{,}4$ V & Ligado\\[2mm]
NMOS & $V_G=0{,}2$ V; $V_S=0$ V & $V_{GS}=0{,}2<0{,}4$ V & Cortado\\[2mm]
PMOS & $V_S=1{,}2$ V; $V_G=0$ V & $V_{SG}=1{,}2>0{,}4$ V & Ligado\\[2mm]
PMOS & $V_S=1{,}2$ V; $V_G=1{,}2$ V & $V_{SG}=0<0{,}4$ V & Cortado\\\bottomrule\end{tabular}\end{adjustbox}
\vspace{3mm}\pergunta{E um NMOS com $V_G=1{,}2$ V e $V_S=1{,}0$ V?}
\fonte{Valores didáticos: $V_{TN}=|V_{TP}|=0{,}4$ V; não são parâmetros de um PDK ou do laboratório.}
\end{frame}
\note{VGS=0,2 V: cortado no modelo, apesar de porta em 1,2 V. O limiar varia com tecnologia, corpo e condições de operação.}

\begin{frame}{Modelo de chave: útil, mas limitado}
\duas{\topico{O que o modelo prevê}{\begin{itemize}\item Existência de caminhos condutores.\item Valor lógico estável esperado.\item Composição de redes de transistores.\end{itemize}}}{\topico{O que exige um modelo elétrico}{\begin{itemize}\item Corrente e resistência de condução.\item Velocidade de carga e descarga.\item Fugas e comportamento próximo ao limiar.\end{itemize}}}
\ideia{Ligado não significa resistência zero. Cortado não significa corrente rigorosamente nula.}
\end{frame}
\note{Se perguntado: corte, triodo e saturação são regimes elétricos. MOSFET em lógica percorre regimes na transição. Saturação de MOS não é sinônimo de chave ideal fechada.}

\begin{frame}{Comprimento e largura: as dimensões do MOSFET}
\small

\duas{\topico{Comprimento $L$}{Distância do canal entre fonte e dreno. Participa do controle da corrente e do comportamento do dispositivo.}
\topico{Largura $W$}{Extensão transversal do canal. Aumentar $W$ tende a ampliar a capacidade de fornecer corrente.}}{\topico{Uma comparação inicial}{Para condições e tecnologia semelhantes, aumentar $W/L$ tende a aumentar a condução.}
\topico{Há um compromisso}{Um dispositivo maior também costuma apresentar capacitâncias e área maiores.}}
\ideia{Mais corrente pode acelerar a carga da saída, mas o dimensionamento também muda a carga que o transistor impõe ao estágio anterior.}
\end{frame}
\note{Qualificar todas as comparações: tecnologia, polarização e regime semelhantes. Não aplicar proporcionalidades de canal longo como leis exatas de processos avançados. O objetivo é introduzir a existência de dimensionamento, não escolher dimensões do laboratório.}

\begin{frame}{Ligado não é uma única região de operação}

\begin{description}
\item[Corte] No modelo inicial, não há canal forte e a condução é desprezada.
\item[Região triodo] Com canal formado e pequena tensão entre dreno e fonte, o dispositivo pode ser aproximado por uma resistência controlada.
\item[Saturação] A corrente fica menos sensível a $V_{DS}$ no modelo de canal longo; é uma região importante em circuitos analógicos.
\end{description}
\topico{Durante uma comutação}{Um MOSFET digital pode atravessar diferentes regiões enquanto a tensão da saída muda. O modelo de chave resume a conectividade, sem calcular toda essa trajetória.}
\end{frame}
\note{Não é necessário memorizar equações nesta introdução. Caso solicitado, para NMOS de canal longo com VGS acima do limiar: triodo para VDS menor que VGS menos VTN e saturação para VDS maior ou igual a essa diferença. Modelo qualitativo; omite modulação de canal e demais efeitos.}



\section{CMOS e integração}
\begin{frame}{Por que usar os dois tipos?}
\centering\begin{adjustbox}{max width=\linewidth}\begin{tabular}{ccll}\toprule Entrada & NMOS & PMOS & Papel no CMOS\\\midrule
Baixa & Cortado & Ligado & PMOS pode levar a saída a $\VDD$\\[2mm]
Alta & Ligado & Cortado & NMOS pode levar a saída a GND\\\bottomrule\end{tabular}\end{adjustbox}
\vspace{4mm}\ideia{CMOS significa MOS complementar: NMOS e PMOS trabalham em redes complementares.}
\fonte{Para o inversor, com fontes em GND e $\VDD$ e entradas próximas dos trilhos.}
\end{frame}
\note{CMOS não é um terceiro tipo de transistor. Os corpos permanecem ligados aos trilhos no modelo adotado.}

\begin{frame}{Primeiro circuito: um inversor CMOS}
\duas{\centering\inversor}{\vspace{2mm}\begin{adjustbox}{max width=\linewidth}\begin{tabular}{cc}\toprule Entrada $A$ & Saída $Y$\\\midrule 0 & 1\\1 & 0\\\bottomrule\end{tabular}\end{adjustbox}
\vspace{3mm}\[Y=\overline{A}\]
\topico{A mesma entrada controla os dois}{Um transistor carrega a saída; o outro a descarrega.}}
\fonte{Corpos omitidos; NMOS em GND e PMOS em $\VDD$. Desenvolvimento completo na aula 2.}
\end{frame}
\note{Pedir que a turma trace o caminho de subida para A=0 e o de descida para A=1. Curvas e consumo ficam para a aula 2.}

\begin{frame}{Uma transição completa do inversor}

\begin{enumerate}
\item \textbf{Antes: $A=0$.} PMOS conduz; a capacitância de saída está carregada e $Y$ está próximo de $\VDD$.
\item \textbf{A entrada sobe.} O NMOS passa a conduzir e o PMOS deixa de conduzir ao longo da transição.
\item \textbf{A carga sai do nó.} A capacitância se descarrega pela rede NMOS; $Y$ desce com atraso.
\item \textbf{Depois: $A=1$.} A saída fica próxima de GND. Quando a entrada voltar a zero, o PMOS recarregará a saída.
\end{enumerate}
\ideia{Uma operação lógica tem uma realização física: movimentar carga entre nós e alimentação.}
\end{frame}
\note{Pode haver condução simultânea temporária durante a transição. A capacitância reúne contribuições dos dispositivos, fios e entradas seguintes. Na aula 2, essa descrição será conectada a atraso e energia.}


\begin{frame}{Uma chave aberta não produz automaticamente um zero}
\duas{\topico{Nó conectado}{Há um caminho ativo até uma referência. A tensão tende ao trilho correspondente.}}{\topico{Nó flutuante}{Não há caminho ativo até um trilho. A tensão depende de carga anterior, capacitâncias e fugas.}}
\vspace{4mm}\pergunta{Se retirarmos o PMOS do inversor, quem recarrega a saída quando o NMOS corta?}
\end{frame}
\note{Sem outro elemento, ninguém garante a recarga; nó flutuante. Não confundir alta impedância com nível lógico 0.}

\begin{frame}{Do dispositivo ao circuito integrado}
\centering\begin{adjustbox}{max width=\linewidth}\begin{tikzpicture}[node distance=4mm]
\node[caixa,text width=2.6cm] (a) {\textbf{Dispositivo}\\NMOS / PMOS};
\node[caixa,text width=2.6cm,right=of a] (b) {\textbf{Célula lógica}\\Inversor / NAND};
\node[caixa,text width=2.6cm,right=of b] (c) {\textbf{Bloco}\\Somador / memória};
\node[caixa,text width=2.6cm,right=of c] (d) {\textbf{Sistema}\\Processador / SoC};
\draw[seta](a)--(b);\draw[seta](b)--(c);\draw[seta](c)--(d);
\end{tikzpicture}\end{adjustbox}
\vspace{5mm}\begin{itemize}\item O esquema descreve conexões e funções.
\item O layout descreve geometrias e camadas físicas, incluindo contatos e interconexões entre dispositivos.
\item A fabricação materializa os dispositivos e suas interconexões.\end{itemize}
\end{frame}
\note{SoC: sistema em um chip. Memórias e blocos analógicos podem usar estruturas próprias; nem todo circuito é sintetizado a partir de portas digitais.}

\begin{frame}{Como o silício vira um circuito?}
\begin{enumerate}\item \textbf{Estruturas no material:} regiões dopadas, isolação e porta.
\item \textbf{Contatos e metais:} conexões entre terminais.
\item \textbf{Máscaras e processos:} repetição de etapas para definir as camadas.
\item \textbf{Teste e encapsulamento:} acesso elétrico e integração ao produto.\end{enumerate}
\vspace{2mm}\ideia{No chip, os transistores são formados no processo de fabricação. O projetista define geometrias e conexões compatíveis com esse processo.}
\end{frame}
\note{Diferenciar esquema, layout e chip. PDK: conjunto de modelos e regras de uma tecnologia; detalhes ficam para a aula prática.}

\begin{frame}{Esquema, layout e chip: o que precisa coincidir?}

\topico{Esquema elétrico}{Define dispositivos, parâmetros e conexões; permite prever o comportamento por modelos.}
\topico{Layout}{Organiza geometrias, contatos e interconexões nas camadas de uma tecnologia. Precisa respeitar regras de fabricação.}
\begin{itemize}
\item \textbf{DRC:} verifica regras geométricas do processo.
\item \textbf{LVS:} compara a conectividade extraída do layout com o esquema.
\item \textbf{Extração:} estima elementos parasitas para analisar seu impacto no circuito.
\end{itemize}
\end{frame}
\note{Definir as siglas apenas como visão do fluxo. Não elaborar a aula prática: nenhum comando de Virtuoso, escolha de PDK ou dimensão está sendo definido. LVS é layout versus schematic; DRC é design rule check. A extração de parasitas é usualmente chamada PEX.}

Considere $\VDD=1{,}2$ V e $V_{TN}=|V_{TP}|=0{,}4$ V, apenas para este exercício.
\vspace{3mm}\begin{enumerate}\item NMOS: $V_G=0$ V e $V_S=0$ V.
\item PMOS: $V_G=0{,}2$ V e $V_S=1{,}2$ V.
\item NMOS: $V_G=1{,}2$ V e $V_S=1{,}0$ V.
\item No inversor, $A=0$: qual caminho está ativo e qual é $Y$?\end{enumerate}

\end{frame}
