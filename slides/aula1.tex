\section{Junção PN}
\begin{frame}{A região de depleção nasce do contato}
\only<1|handout:0>{\figura{\juncaoPN{0}}\etapa{1/3}{Instante do contato: concentrações diferentes.}}
\only<2|handout:0>{\figura{\juncaoPN{1}}\etapa{2/3}{Portadores difundem e se recombinam.}}
\only<3|handout:1>{\figura{\juncaoPN{2}}\etapa{3/3}{Íons expostos criam o campo interno.}}
\end{frame}
\note{Etapa 1: pedir a direção da difusão dos elétrons e das lacunas. Etapa 2: discutir por que a região perto da interface perde portadores móveis. Etapa 3: identificar íons negativos no lado P e positivos no lado N. O campo interno aponta de N para P. Em equilíbrio, deriva e difusão se compensam; depleção não significa ausência de toda carga. Desenhos sem escala; só majoritários são representados fora da faixa cinza.}

\begin{frame}{Uma tensão externa muda a depleção}
\only<1|handout:0>{\figura{\juncaoPN{3}}\etapa{1/2}{Direta: P positivo em relação a N.}}
\only<2|handout:1>{\figura{\juncaoPN{4}}\etapa{2/2}{Direta estreita; reversa alarga a depleção.}}
\end{frame}
\note{Pedir para comparar a largura cinza entre os dois cliques. Em direta a barreira diminui e a corrente pode crescer fortemente; em reversa, há pequena fuga antes da ruptura. A seta mostra a direção do campo, não seu valor. Não fixar 0,7 V como tensão universal de condução. O material para consulta retoma os dois estados.}

\begin{frame}{Da junção PN ao diodo}
\figura{\begin{circuitikz}[scale=1.5]\draw (0,0) node[left]{P · ânodo} to[D] (3,0) node[right]{N · cátodo};\draw[-{Latex},blue_theme,thick] (.8,-.7)--node[below]{corrente convencional em direta}(2.2,-.7);\end{circuitikz}}
\chamada{A tensão externa altera a barreira da junção.}
\small\centering Direta: a corrente cresce fortemente.\quad Reversa: pequena fuga antes da ruptura.
\end{frame}
\note{O diodo é uma junção PN com terminais acessíveis. Em direta, P está positivo em relação a N e a corrente convencional vai do ânodo ao cátodo. A barra do símbolo marca o cátodo. Não apresentar 0,7 V como constante universal: corrente e tensão dependem do dispositivo e das condições.}

\section{MOSFET}
\begin{frame}{Conheça os terminais do MOSFET}
\figura{\moscanal{0}}
\chamada{Porta controla; fonte e dreno conduzem pelo canal.}
\end{frame}
\note{Explicar G, S, D e corpo B. MOS: metal--óxido--semicondutor; FET: transistor de efeito de campo. O desenho é de um NMOS planar de enriquecimento. O corpo é do tipo P e as difusões, n+. O dielétrico separa eletricamente a porta do canal; não há fio da porta para dentro do semicondutor.}

\begin{frame}{Ativar o MOSFET: acompanhe a superfície}
\only<1|handout:0>{\figura{\moscanal{0}}\etapa{1/4}{Sem canal forte: chave ideal aberta.}}
\only<2|handout:0>{\figura{\moscanal{1}}\etapa{2/4}{O campo de porta afasta lacunas.}}
\only<3|handout:0>{\figura{\moscanal{2}}\etapa{3/4}{Forma-se um canal de inversão.}}
\only<4|handout:1>{\figura{\moscanal{3}}\etapa{4/4}{Uma diferença entre dreno e fonte produz corrente.}}
\end{frame}
\note{Fazer uma pausa a cada clique e pedir a mudança observada. A superfície P sofre depleção e depois inversão; o corpo não vira inteiro um material N. Os elétrons do canal não atravessam o dielétrico da porta. Na última etapa, a corrente convencional vai do dreno à fonte para VD maior que VS; os elétrons se deslocam no sentido oposto. O limiar é uma referência de inversão forte, e não uma fronteira de corrente exatamente nula.}

\begin{frame}{Duas chaves complementares}
\duas{\centering\textcolor{blue_theme}{\large\textbf{NMOS}}\par
\figura{\begin{circuitikz}\node[nmos] (n) at (0,0) {};\draw (n.G) node[left]{G};\draw(n.S) node[below]{S};\draw(n.D) node[above]{D};\end{circuitikz}}
\[V_{GS}=V_G-V_S\]
Conduz se $V_{GS}>V_{TN}$.}{\centering\textcolor{blue_theme}{\large\textbf{PMOS}}\par
\figura{\begin{circuitikz}\node[pmos] (p) at (0,0) {};\draw(p.G) node[left]{G};\draw(p.S) node[above]{S};\draw(p.D) node[below]{D};\end{circuitikz}}
\[V_{SG}=V_S-V_G\]
Conduz se $V_{SG}>|V_{TP}|$.}
\chamada{Compare terminais, não apenas a tensão de porta.}
\end{frame}
\note{Modelo ideal de chave para MOSFETs de enriquecimento. No inversor, a fonte NMOS fica em GND e a fonte PMOS em VDD; assim, entrada alta liga NMOS e entrada baixa liga PMOS. Em redes com fontes elevadas a regra precisa de mais cuidado. Correntes de sub-limiar e demais efeitos ficam nas notas.}

\begin{frame}{Qual chave está ligada?}
\centering\small
\only<1|handout:0>{\begin{tabular}{llll}\toprule Tipo & Porta & Fonte & Estado\\\midrule NMOS&0 V&0 V&?\\PMOS&0,2 V&1,2 V&?\\NMOS&1,2 V&1,0 V&?\\\bottomrule\end{tabular}\par\bigskip Calcule $V_{GS}$ ou $V_{SG}$ antes de responder.}
\only<2|handout:1>{\begin{tabular}{llll}\toprule Tipo & Diferença & Comparação & Estado\\\midrule NMOS&$V_{GS}=0$ V&$0<0{,}4$&Cortado\\PMOS&$V_{SG}=1{,}0$ V&$1{,}0>0{,}4$&Ligado\\NMOS&$V_{GS}=0{,}2$ V&$0{,}2<0{,}4$&Cortado\\\bottomrule\end{tabular}\chamada{Porta alta não basta se a fonte também estiver alta.}}
\fonte{Valores didáticos: limiares constantes $V_{TN}=|V_{TP}|=0{,}4$ V. Não são parâmetros do laboratório.}
\end{frame}
\note{Usar 5 minutos de resolução em duplas e revelar a correção. A terceira linha é o diagnóstico central: VG=1,2 V com VS=1 V resulta em VGS de apenas 0,2 V. Nos dispositivos reais, o limiar também pode depender do corpo e das condições de operação.}

\begin{frame}{Três detalhes que o desenho simplifica}
\figura{\begin{tikzpicture}[x=1cm,y=1cm]
\draw[blue_theme,fill=blue_theme!8] (0,0) rectangle (8,2.4);\node at (4,1.2){canal};\node at(1.25,1.2){S};\node at(6.75,1.2){D};
\fill[yellow!40,draw=blue_theme] (.5,0) rectangle (2,2.4);\fill[yellow!40,draw=blue_theme] (6,0) rectangle (7.5,2.4);
\draw[<->,blue_theme] (2,2.9)--node[above]{$L$}(6,2.9);\draw[<->,blue_theme] (8.4,0)--node[right]{$W$}(8.4,2.4);
\node[font=\scriptsize] at (4,-.5){Vista superior esquemática};
\end{tikzpicture}}
\begin{itemize}\item Corpo NMOS em GND; corpo PMOS em $\VDD$ no modelo bulk.
\item Dimensões $W$ e $L$ afetam corrente, carga e área.
\item Uma chave real tem resistência e correntes de fuga.\end{itemize}
\end{frame}
\note{A vista superior é esquemática: L mede o percurso entre fonte e dreno; W mede a extensão transversal do canal. As junções PN entre corpo e difusões justificam as conexões de corpo. A fonte de um transistor empilhado não precisa coincidir com o corpo. Não derivar modelos de corrente ou regimes neste encontro.}

\begin{frame}{A saída armazena carga}
\duas{\figura{\begin{circuitikz}\draw(0,2) node[above]{$Y$} to[C,l=$C_L$] (0,0) node[ground]{};\end{circuitikz}}}{\vspace{8mm}\[Q=C_LV_Y\]\par\medskip

Descarregar também.\par\medskip
Chave aberta não garante tensão zero.}
\end{frame}
\note{A capacitância de carga é um equivalente dos dispositivos, entradas seguintes e fios. Se ambos os caminhos forem retirados, a saída pode ficar flutuante e depender da carga anterior e das fugas. Retomar a pergunta sobre corrente zero não significar tensão zero. A aula 2 explorará o atraso com formas de onda.}
