\section{Trilhas de estudo}
\begin{frame}{Digital · transformar funções em hardware}
\duas{\topico{Pergunta central}{Como implementar uma função correta respeitando desempenho, consumo e área?}
\topico{O que se projeta}{Somadores, controladores, interfaces, memórias e aceleradores.}}{\topico{Sequência de estudo}{Lógica combinacional $\rightarrow$ registradores e máquinas de estados $\rightarrow$ HDL/RTL $\rightarrow$ verificação $\rightarrow$ síntese e análise temporal.}
\topico{Primeiro projeto}{Uma ULA de 4 bits, com operações aritméticas e lógicas e testes automáticos.}}
\fonte{HDL: linguagem de descrição de hardware. RTL: descrição por registradores e transferências. ULA: unidade lógica e aritmética.}
\end{frame}
\note{Conectar a ULA às NAND/NOR já estudadas. FPGA pode permitir prototipagem digital; um ASIC exige ainda etapas de implementação e fabricação próprias. Não tratar escrever HDL como escrever um programa sequencial comum.}

\begin{frame}{Arquitetura de computadores · organizar a computação}
\small
\duas{\topico{Pergunta central}{Como organizar unidades de execução, memória e comunicação para executar programas com eficiência?}
\topico{O que se estuda}{Conjunto de instruções, caminho de dados, controle, pipeline, caches e paralelismo.}}{\topico{Sequência de estudo}{Lógica sequencial $\rightarrow$ ULA e registradores $\rightarrow$ processador simples $\rightarrow$ memória $\rightarrow$ desempenho e aceleradores.}
\topico{Primeiro projeto}{Construir um caminho de dados simples e medir ciclos necessários para executar uma pequena tarefa.}}
\ideia{Arquitetura define escolhas do sistema; microarquitetura e RTL aproximam essas escolhas de uma implementação.}
\end{frame}
\note{Distinguir ISA: contrato de instruções visível ao software; microarquitetura: organização que o implementa. O exemplo do acelerador de Luan ilustra a ponte com digital. É um desdobramento de estudo, não conteúdo exigido para concluir o curso.}

\begin{frame}{Analógica · trabalhar com sinais contínuos}
\small
\duas{\topico{Pergunta central}{Como amplificar, filtrar e regular sinais preservando precisão, faixa útil e estabilidade?}
\topico{O que se projeta}{Amplificadores, referências, espelhos de corrente, filtros e reguladores.}}{\topico{Sequência de estudo}{Circuitos $\rightarrow$ polarização do MOSFET $\rightarrow$ pequenos sinais $\rightarrow$ realimentação $\rightarrow$ ruído, variações e layout.}
\topico{Primeiro projeto}{Amplificador fonte comum: definir polarização e comparar ganho e excursão de saída.}}
\ideia{O mesmo MOSFET deixa de ser apenas uma chave e passa a controlar corrente de forma contínua.}
\end{frame}
\note{Não afirmar que analógico significa sempre operação na região triodo. Amplificadores frequentemente usam transistores em saturação, dependendo da topologia. O trabalho de Luiz com amplificador operacional é exemplo de desdobramento.}

\begin{frame}{Sinais mistos · ligar o mundo físico ao digital}
\duas{\topico{Pergunta central}{Como converter sinais e coordenar blocos analógicos e digitais?}
\topico{O que se projeta}{Conversores, interfaces de sensores, sistemas de aquisição e blocos de sincronização.}}{\topico{Sequência de estudo}{Base analógica + digital $\rightarrow$ amostragem $\rightarrow$ quantização $\rightarrow$ conversores $\rightarrow$ integração e verificação conjunta.}
\topico{Primeiro projeto}{Modelo de conversor de 3 bits: relacionar tensão de entrada, código digital e erro de quantização.}}
\fonte{\textit{Mixed-signal}: sinais mistos. ADC: conversor analógico--digital. DAC: conversor digital--analógico.}
\end{frame}
\note{Idealização para projeto inicial: faixas uniformes de quantização. Integração também exige atenção a alimentação, referência, sincronização e interferência. Projeto de conversor transistor a transistor pode vir depois.}

\begin{frame}{RF integrada · transmitir e receber sinais}
\small
\duas{\topico{Pergunta central}{Como receber e transmitir sinais em radiofrequência com ganho, baixo ruído e boa linearidade?}
\topico{O que se projeta}{Amplificadores de baixo ruído, misturadores, osciladores e transmissores/receptores integrados.}}{\topico{Sequência de estudo}{Analógica + sinais e sistemas $\rightarrow$ eletromagnetismo $\rightarrow$ impedâncias e parâmetros S $\rightarrow$ ruído e circuitos de RF.}
\topico{Primeiro projeto}{Rede LC de casamento de impedâncias: observar a resposta em frequência e o efeito das perdas.}}
\ideia{Em RF, interconexões, encapsulamento e componentes parasitas têm papel central no comportamento do circuito.}
\end{frame}
\note{LC: indutor e capacitor. Parâmetros S descrevem ondas incidentes e refletidas em portas. RF é um aprofundamento conectado à analógica, sinais e eletromagnetismo. Evitar definir RF por uma frequência de corte universal.}
